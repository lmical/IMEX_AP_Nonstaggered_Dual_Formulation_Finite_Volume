\begin{frame}
	\frametitle{Governing equations}

		\begin{equation*}
			\label{sys}
			\frac{\partial}{\partial t}\uvec{U}(\uvec{x},t)+div_{\uvec{x}} \uvec{F}(\uvec{U}(\uvec{x},t))=\boldsymbol{0}, \quad (\uvec{x},t) \in \Omega\times \mathbb{R}^+_0,
		\end{equation*}
		\visible<2->{
			\begin{itemize}
				\item $\Omega\subseteq \mathbb{R}^D$ spatial domain, a connected open set}
				\visible<3->{\item $\uvec{U}:\overline{\Omega} \times \mathbb{R}^+_0 \rightarrow \mathcal{U} \subseteq \mathbb{R}^Q$ unknown solution} %with $N \in \mathbb{N}$ number of scalar equations of the system and $\mathcal{U}$ space of adimssible states
				\visible<4->{\item $\uvec{F}:\mathcal{U} \rightarrow \mathbb{R}^{Q \times D}$ flux}
				\visible<5->{\item $\uvec{F}(\uvec{U})=(\uvec{F}_1,\uvec{F}_2,\dots,\uvec{F}_D) \in \mathbb{R}^{Q \times D}$ with $div_{\uvec{x}} \uvec{F}(\uvec{U})=\sum\limits_{d=1}^D \frac{\partial \uvec{F}_{d}}{\partial x_d} (\uvec{U}) \in \mathbb{R}^{Q}$} 
				\visible<6->{\item $\sum\limits_{d=1}^D \omega_d \frac{\partial \uvec{F}_{d}}{\partial \uvec{U}} (\uvec{U})$ real diagonalizable $\forall \uvec{U}\in \mathcal{U}$, $\forall \uvec{\omega}\in \mathbb{R}^D$}
				\visible<7->{\item with some initial $\uvec{U}_0=\uvec{U}(\uvec{x},0)$ and boundary conditions}
			\end{itemize}
\end{frame}

\begin{frame}[label=BalanceLawsSomeExamples]
	\frametitle{An example: Euler equations}
	$$\frac{\partial}{\partial t}\uvec{U}+div_{\uvec{x}} \uvec{F}(\uvec{U})=\boldsymbol{0},$$
	
	$$
	\uvec{U}=\begin{pmatrix}
		\rho\\
		\rho u\\
		\rho v\\
		\rho w\\
		E
	\end{pmatrix}, \quad \uvec{F}(\uvec{U})=\begin{pmatrix}
		\rho u & \rho v & \rho w \\ 
		\rho u^2+p & \rho u v & \rho u w \\
		\rho u v & \rho v^2+p & \rho v w \\
		\rho u w & \rho v w & \rho w^2+p \\
		(E+p)u & (E+p)v & (E+p)w
	\end{pmatrix},
	$$
	with $p=(\gamma -1) \left[E- \frac{1}{2}\rho (u^2+v^2+w^2) \right]$.
	
\end{frame}

\begin{frame}[label=BalanceLawsDerivation]
	\frametitle{Derivation}
	\begin{columns}

		\end{column}
		\begin{column}{0.70\textwidth}
			\begin{itemize}
				\item $V\subseteq \Omega$ control volume
				\item $\uvec{U}$ conserved quantities
				\item $\flux \cdot \uvec{\nu}$ exiting from the control volume
			\end{itemize}
		\end{column}
	\end{columns}
	\visible<2->{		\scalebox{0.7}{\begin{minipage}[t]{1.2\textwidth}
				\begin{beamercolorbox}[sep=0.5em,wd=\textwidth]{blockcolor1} 
					$$\frac{\partial}{\partial t} \int_V \uvec{U} d \uvec{x} = - \int_{\partial V} \flux \cdot \uvec{\nu} d \sigma(\uvec{x}) , \quad  \forall V\subseteq \Omega.
					$$
					$$\Downarrow$$
					$$\int_V \frac{\partial}{\partial t}\uvec{U} d \uvec{x} = - \int_V div_{\uvec{x}}\flux d \uvec{x} , \quad  \forall V\subseteq \Omega.$$
					$$\Downarrow$$
					$$\int_V \left(\frac{\partial}{\partial t}\uvec{U}  + div_{\uvec{x}}\flux  \right) d \uvec{x}=\uvec{0}, \quad  \forall V\subseteq \Omega.$$
				\end{beamercolorbox}
	\end{minipage}} }
\end{frame}


\begin{frame}
	\frametitle{A structural property: Conservation}
		\begin{equation*}
			\frac{\partial}{\partial t}\uvec{U}+ \frac{\partial}{\partial x} \uvec{F}(\uvec{U})=\boldsymbol{0}, \quad (x,t) \in \Omega\times \mathbb{R}^+_0
		\end{equation*}
		$$\Downarrow$$
$$\int_\Omega \left( \frac{\partial}{\partial t}\uvec{U}+ \frac{\partial}{\partial x} \uvec{F}(\uvec{U}) \right) d \uvec{x} =\boldsymbol{0}$$
		$$\Downarrow$$
$$\frac{d}{d t} \int_\Omega \uvec{U} d \uvec{x} = - \left[\uvec{F}(\uvec{U})\right]_{x=x_L}^{x=x_R}   =\boldsymbol{0}$$
\end{frame}

\begin{frame}
	\frametitle{Alternative (primitive) formulations}
	
	\centering
	
	{\footnotesize
	\begin{align*}
		\frac{\partial}{\partial t}\textcolor{black}{\uvec{U}}+&\frac{\partial}{\partial x}\uvec{F}(\uvec{U})=\uvec{0}\\
		\textcolor{black}{\uvec{U}:=\begin{pmatrix}\rho\\\rho v\\E\end{pmatrix}},\quad
		\uvec{F}(\uvec{U}):=&\begin{pmatrix}\rho v\\\rho v^2+p\\v(E+p)\end{pmatrix}, \quad E=\frac{p}{\gamma-1}+\frac{1}{2}\rho u^2
	\end{align*}
	
	
	\begin{align*}
		\frac{\partial}{\partial t}&\textcolor{black}{\uvec{V}} + \frac{\partial}{\partial x}\uvec{\widetilde F}(\uvec{V}) = \textcolor{black}{\B(\uvec{V})\frac{\partial}{\partial x}\uvec{V}} \\
		\textcolor{black}{\uvec{V} :=} \textcolor{black}{\begin{pmatrix} \rho \\ v \\ p \end{pmatrix}}, \quad
		\uvec{\widetilde F}(\uvec{V})& := \begin{pmatrix} \rho v \\ \dfrac{v^2}{2} \\ p v \end{pmatrix}, \quad \textcolor{black}{\B(\uvec{V}) :=} \textcolor{black}{\begin{pmatrix}
				0 & 0 & 0 \\
				0 & 0 & -\dfrac{1}{\rho} \\
				0 & -(\gamma-1) p & 0
		\end{pmatrix}}
	\end{align*}
	}

	\visible<2->{
		{\footnotesize		
			\centering
			...offers some advantages:
			\begin{itemize}
				\item Low-Mach ($p$=const, $div_{\uvec{x}}\uvec{v}=0$)
				\item Multifluid ($p,v$=const at interface)
				\item Positivity preservation ($p > 0$, not $E$)
				\item One measures primitive variables
				\item etc...
			\end{itemize}
		}
	}

\begin{frame}
	\frametitle{DF-FV, post--processing {\small $\left(\overline{\uvec{U}}_i^*,\overline{\uvec{V}}_\iph^* \right) \xrightarrow{\text{PP}} \left(\overline{\uvec{U}}_i^{n+1}, \overline{\uvec{V}}_\iph^{n+1}\right)$ }}

{\tiny
	
{\bf Step 1.} Compute the conserved variables at $x=x_\iph$
$$
\bm U_\iph^*:=\bm U\big(\xbar{\bm V}_\iph^{\,*}\big).
$$

\vskip3pt
\noindent
{\bf Step 2.} Perform the piecewise linear reconstruction for $\bm U$ 
\begin{equation*}
	\bm U_\iph^{*,-}:=\,\xbar{\bm U}_i^{\,*}+\frac{\dx}{2}(\bm U_x)_i^*,\quad
	\bm U_\iph^{*,+}:=\,\xbar{\bm U}_{i+1}^{\,*}-\frac{\dx}{2}(\bm U_x)_{i+1}^*,\quad (\bm U_x)_i^*=2\,{\rm minmod}\left(\frac{\,\xbar{\bm U}_i^{\,*}-\bm U_\imh^*}{\dx},\frac{\bm U_\iph^*-\,\xbar{\bm U}_i^{\,*}}{\dx}\right).
\end{equation*}


\vskip3pt
\noindent
{\bf Step 3.} Correct the primitive variables
\begin{align*}
	\bm U_\iph^{**}&:=\hf\Big(\bm U_\iph^{*,-}+\bm U_\iph^{*,+}\Big),\\
	\xbar{\bm V}_\iph^{n+1}&:=\bm V\big(\bm U_\iph^{**}\big).
\end{align*}



\vskip3pt
\noindent
{\bf Step 4.} Correct the conserved variables
\begin{equation*}
	\xbar{\bm U}_i^{n+1}=\hf\Big(\bm U_\imh^{**}+\bm U_\iph^{**}\Big).
	\label{2.9}
\end{equation*}

}

\end{frame}
